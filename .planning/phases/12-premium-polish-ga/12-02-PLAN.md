    1. Create `lib/core/services/observability_service.dart`:
    ```dart
    import 'package:firebase_analytics/firebase_analytics.dart';
    import 'package:firebase_crashlytics/firebase_crashlytics.dart';
    import 'package:flutter/foundation.dart';

    /// Single facade for Crashlytics, Analytics, and Performance.
    /// All observability calls in the app go through this service.
    class ObservabilityService {
      ObservabilityService._();

      static final FirebaseCrashlytics _crashlytics = FirebaseCrashlytics.instance;
      static final FirebaseAnalytics _analytics = FirebaseAnalytics.instance;

      /// Wire into FlutterError.onError (fatal Flutter framework errors).
      static void onFlutterError(FlutterErrorDetails details) {
        _crashlytics.recordFlutterFatalError(details);
      }

      /// Record a non-fatal exception with optional context keys.
      static Future<void> recordNonFatal(
        Object exception,
        StackTrace? stack, {
        Map<String, String> context = const {},
      }) async {
        for (final entry in context.entries) {
          await _crashlytics.setCustomKey(entry.key, entry.value);
        }
        await _crashlytics.recordError(exception, stack, fatal: false);
      }

      /// Log an analytics event. Parameters are String or int values only (Firebase limit).
      static Future<void> logEvent(
        String name, {
        Map<String, Object>? parameters,
      }) async {
        // In debug/profile, skip to avoid polluting analytics with dev data.
        if (kDebugMode) return;
        await _analytics.logEvent(name: name, parameters: parameters);
      }

      /// Set the analytics + crashlytics user identifier (call after sign-in).
      static Future<void> setUserId(String userId) async {
        await _analytics.setUserId(id: userId);
        await _crashlytics.setUserIdentifier(userId);
      }

      /// Clear user identifier on sign-out.
      static Future<void> clearUserId() async {
        await _analytics.setUserId(id: null);
        await _crashlytics.setUserIdentifier('');
      }
    }
    ```

    2. Update `lib/main.dart`:
    In the `main()` async function, AFTER `await Firebase.initializeApp(...)` and BEFORE `runApp(...)`, add:

    ```dart
      FirebaseCrashlytics.instance.recordError(error, stack, fatal: true);
      return true;
    };
    ```
    Import `observability_service.dart` and `firebase_crashlytics` in main.dart.

    Also call `ObservabilityService.setUserId(user.id)` in the auth state listener (already in main.dart or AuthWrapper — add after sign-in confirmation; call `clearUserId()` on sign-out).
