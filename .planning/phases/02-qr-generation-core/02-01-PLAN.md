   class EventData extends QRDataModel {
     final String title;
     final String location;
     final DateTime startDateTime;
     final DateTime? endDateTime;
     final String? description;

     EventData({
       required this.title,
       required this.location,
       required this.startDateTime,
       this.endDateTime,
       this.description,
     });

     String _formatDt(DateTime dt) {
       // Format as YYYYMMDDTHHMMSS (local, no Z suffix — avoids timezone complexity for v1)
       final y = dt.year.toString().padLeft(4, '0');
       final mo = dt.month.toString().padLeft(2, '0');
       final d = dt.day.toString().padLeft(2, '0');
       final h = dt.hour.toString().padLeft(2, '0');
       final mi = dt.minute.toString().padLeft(2, '0');
       final s = dt.second.toString().padLeft(2, '0');
       return '$y$mo${d}T$h$mi$s';
     }

     @override
     String get qrData {
       final buf = StringBuffer('BEGIN:VCALENDAR\nVERSION:2.0\nBEGIN:VEVENT\n');
       buf.write('SUMMARY:$title\n');
       buf.write('DTSTART:${_formatDt(startDateTime)}\n');
       if (endDateTime != null) {
         buf.write('DTEND:${_formatDt(endDateTime!)}\n');
       }
       if (location.isNotEmpty) buf.write('LOCATION:$location\n');
       if (description != null && description!.isNotEmpty) {
         buf.write('DESCRIPTION:$description\n');
       }
       buf.write('END:VEVENT\nEND:VCALENDAR');
       return buf.toString();
     }

     @override
     String get displayText {
       final date = '${startDateTime.day}/${startDateTime.month}/${startDateTime.year}';
       return '$title — $date${location.isNotEmpty ? ', $location' : ''}';
     }

     @override
     List<Object?> get props => [title, location, startDateTime, endDateTime, description];
   }
   ```

   @override
   Future<List<QRCodeEntity>> getUserQRCodesPaginated(
     String userId,
     int page,
     int pageSize,
   ) async {
     final from = page * pageSize;
     final to = from + pageSize - 1;
     final response = await client
         .from('qr_codes')
         .select()
         .eq('user_id', userId)
         .order('created_at', ascending: false)
         .range(from, to);
     return (response as List)
         .map((json) => QRCodeModel.fromJson(json as Map<String, dynamic>))
         .toList();
   }
   ```
   Check what import `QRCodeModel` uses in this file — it should already be imported. If `response` is typed differently (the Supabase SDK returns `List<Map<String, dynamic>>` directly in v2.x), remove the cast and use `response.map(...)` directly.

4. In `qr_repository_impl.dart`, implement the new abstract method:
   ```dart
   @override
   Future<List<QRCodeEntity>> getUserQRCodesPaginated(
     String userId,
     int page,
     int pageSize,
   ) async {
     try {
       return await remoteDataSource.getUserQRCodesPaginated(userId, page, pageSize);
     } catch (e) {
       // Fallback: slice local cache by page
       final cached = await localDataSource.getCachedQRCodes(userId);
       final from = page * pageSize;
       if (from >= cached.length) return <QRCodeEntity>[];
       final to = (from + pageSize).clamp(0, cached.length);
       return cached.sublist(from, to);
     }
   }
   ```
