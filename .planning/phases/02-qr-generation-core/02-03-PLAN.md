```dart
import 'package:flutter_riverpod/flutter_riverpod.dart';
import '../../domain/entities/qr_data_models.dart';

class EventFormState {
  final String title;
  final DateTime? startDateTime;
  final DateTime? endDateTime;
  final String location;
  final String description;
  final bool isValid;

  const EventFormState({
    this.title = '',
    this.startDateTime,
    this.endDateTime,
    this.location = '',
    this.description = '',
    this.isValid = false,
  });

  EventFormState copyWith({...}) { ... }

  EventData? toEventData() {
    if (title.isEmpty || startDateTime == null) return null;
    return EventData(
      title: title,
      location: location,
      startDateTime: startDateTime!,
      endDateTime: endDateTime,
      description: description.isEmpty ? null : description,
    );
  }
}

class EventFormController extends StateNotifier<EventFormState> {
  EventFormController() : super(const EventFormState());

  void updateTitle(String v)       => state = state.copyWith(title: v, isValid: _validate(title: v));
  void updateStartDateTime(DateTime? v) => state = state.copyWith(startDateTime: v, isValid: _validate(startDateTime: v));
  void updateEndDateTime(DateTime? v)   => state = state.copyWith(endDateTime: v);
  void updateLocation(String v)    => state = state.copyWith(location: v);
  void updateDescription(String v) => state = state.copyWith(description: v);

  void loadFromEntity(EventData data, String name) {
    state = EventFormState(
      title: data.title,
      startDateTime: data.startDateTime,
      endDateTime: data.endDateTime,
      location: data.location,
      description: data.description ?? '',
      isValid: true,
    );
  }

  void reset() => state = const EventFormState();

  bool _validate({String? title, DateTime? startDateTime}) {
    final t = title ?? state.title;
    final s = startDateTime ?? state.startDateTime;
    return t.isNotEmpty && s != null;
  }
}

final eventFormProvider = StateNotifierProvider.autoDispose<EventFormController, EventFormState>(
  (ref) => EventFormController(),
);
```
