Create `supabase/functions/wompi-webhook/index.ts`.

This function MUST be deployed with `--no-verify-jwt` because Wompi does NOT send a
Supabase JWT. The function performs its own authentication via SHA256 checksum.

Full implementation:

```typescript
import { createClient } from 'https://esm.sh/@supabase/supabase-js@2';

// Resolve a dot-separated path like "data.transaction.id" against a JSON object.
function resolvePath(obj: unknown, path: string): string {
  return path.split('.').reduce((acc: unknown, key: string) => {
    if (acc && typeof acc === 'object') return (acc as Record<string, unknown>)[key];
    return undefined;
  }, obj)?.toString() ?? '';
}

async function sha256hex(input: string): Promise<string> {
  const encoded = new TextEncoder().encode(input);
  const hashBuffer = await crypto.subtle.digest('SHA-256', encoded);
  return Array.from(new Uint8Array(hashBuffer))
    .map(b => b.toString(16).padStart(2, '0'))
    .join('');
}

Deno.serve(async (req: Request) => {
  if (req.method !== 'POST') {
    return new Response('method not allowed', { status: 405 });
  }

  let body: Record<string, unknown>;
  try {
    body = await req.json();
  } catch {
    return new Response('invalid json', { status: 400 });
  }

  // ── Step 1: Verify checksum ──────────────────────────────────────────────
  const eventsSecret = Deno.env.get('WOMPI_EVENTS_SECRET');
  if (!eventsSecret) {
    console.error('WOMPI_EVENTS_SECRET not set');
    return new Response('server misconfigured', { status: 500 });
  }

  const signature = body.signature as { properties: string[]; checksum: string } | undefined;
  const timestamp = body.timestamp as number | undefined;
  // Wompi may also put checksum in X-Event-Checksum header — check both
  const receivedChecksum = req.headers.get('X-Event-Checksum') ?? signature?.checksum ?? '';

  if (!signature?.properties || !timestamp || !receivedChecksum) {
    console.warn('Missing signature fields; rejecting event');
    return new Response('missing signature', { status: 401 });
  }

  // Build the string to hash: concatenate resolved property values + timestamp + secret
  const propertyValues = signature.properties.map(p => resolvePath(body, p)).join('');
  const toHash = `${propertyValues}${timestamp}${eventsSecret}`;
  const expectedChecksum = await sha256hex(toHash);

  if (expectedChecksum !== receivedChecksum) {
    console.warn('Checksum mismatch; possible forged event');
    return new Response('invalid checksum', { status: 401 });
  }

  // ── Step 2: Idempotency via payment_events.event_id PK ──────────────────
  const eventId = body.id as string | undefined;
  if (!eventId) {
    return new Response('missing event id', { status: 400 });
  }

  const supabase = createClient(
    Deno.env.get('SUPABASE_URL')!,
    Deno.env.get('SUPABASE_SERVICE_ROLE_KEY')!
  );

  const transaction = (body.data as Record<string, unknown>)?.transaction as
    Record<string, unknown> | undefined;
  const transactionRef = transaction?.reference as string | undefined;
  const wompiStatus = transaction?.status as string | undefined;
  const wompiTxId = transaction?.id as string | undefined;
  const paymentMethod = transaction?.payment_method_type as string | undefined;

  // Fetch payment row to get payment_id for the FK
  const { data: paymentRow } = await supabase
    .from('payments')
    .select('id')
    .eq('transaction_ref', transactionRef ?? '')
    .maybeSingle();

  const { error: insertError } = await supabase
    .from('payment_events')
    .insert({
      event_id: eventId,
      payment_id: paymentRow?.id ?? null,
      kind: body.event as string,
      payload: body,
    });

  if (insertError) {
    if (insertError.code === '23505') {
      // Already processed — idempotent response
      console.log(`Event ${eventId} already processed; returning 200`);
      return new Response('already_processed', { status: 200 });
    }
    console.error('payment_events insert error', insertError);
    return new Response('db error', { status: 500 });
  }

  // ── Step 3: Apply FSM transition via RPC ─────────────────────────────────
  if (transactionRef && wompiStatus) {
    const { error: rpcError } = await supabase.rpc('apply_payment_event', {
      p_transaction_ref: transactionRef,
      p_wompi_status: wompiStatus,
      p_wompi_transaction_id: wompiTxId ?? null,
      p_payment_method: paymentMethod ?? null,
      p_raw_payload: body,
    });

    if (rpcError) {
      // Log but return 200 — Wompi must not retry on our FSM errors
      // (duplicate retry would be idempotent anyway, but we don't want noise)
      console.error('apply_payment_event RPC error', rpcError);
    }
  }

  return new Response('ok', { status: 200 });
});
```

Notes:
- `crypto.subtle` is Deno built-in; no import needed for SHA-256.
- Service role key enables writing to payment_events and calling SECURITY DEFINER RPCs.
- The function returns 200 even on RPC errors to prevent Wompi retry storms; the
  event is already idempotently recorded in payment_events for reconciliation.
- Do NOT log the eventsSecret or service_role_key values anywhere.
