CREATE OR REPLACE FUNCTION transition_order(
  p_order_id    UUID,
  p_new_status  order_status
)
RETURNS VOID LANGUAGE plpgsql SECURITY DEFINER AS $$
DECLARE
  v_order   orders%ROWTYPE;
  v_title   TEXT;
  v_body    TEXT;
  v_kind    TEXT;
BEGIN
  -- Load the order (FSM trigger validates the transition on the UPDATE below).
  SELECT * INTO v_order FROM orders WHERE id = p_order_id;
  IF NOT FOUND THEN
    RAISE EXCEPTION 'Order not found: %', p_order_id;
  END IF;

  -- Build notification text based on target state.
  v_kind  := 'order_' || p_new_status::TEXT;
  v_title := CASE p_new_status
    WHEN 'received_by_ops'  THEN 'Orden recibida'
    WHEN 'engraving'        THEN 'Grabando tu QR'
    WHEN 'engraving_failed' THEN 'Reintentando grabado'
    WHEN 'engraving_done'   THEN 'Grabado completado'
    WHEN 'shipped'          THEN '¡Enviado!'
    WHEN 'delivered'        THEN '¡Entregado!'
    WHEN 'cancelled'        THEN 'Orden cancelada'
    WHEN 'refunded'         THEN 'Reembolso procesado'
    ELSE 'Estado actualizado'
  END;
  v_body := CASE p_new_status
    WHEN 'received_by_ops'  THEN 'Recibimos tu orden ' || v_order.order_number || '. ¡Empezamos pronto!'
    WHEN 'engraving'        THEN 'Tu orden ' || v_order.order_number || ' está siendo grabada ahora.'
    WHEN 'engraving_failed' THEN 'Hubo un inconveniente. Estamos rehaciendo tu orden ' || v_order.order_number || '.'
    WHEN 'engraving_done'   THEN 'Tu QR está listo. En proceso de envío.'
    WHEN 'shipped'          THEN 'Tu orden ' || v_order.order_number || ' está en camino.'
    WHEN 'delivered'        THEN 'Tu orden ' || v_order.order_number || ' fue entregada. ¡Disfrútala!'
    WHEN 'cancelled'        THEN 'Tu orden ' || v_order.order_number || ' ha sido cancelada.'
    WHEN 'refunded'         THEN 'El reembolso de tu orden ' || v_order.order_number || ' ha sido procesado.'
    ELSE 'Tu orden ' || v_order.order_number || ' ha cambiado de estado.'
  END;

  -- Update order status (FSM trigger enforces legality).
  UPDATE orders SET status = p_new_status WHERE id = p_order_id;

  -- Insert notification row (triggers DB webhook → send-fcm).
  PERFORM insert_order_notification(
    v_order.user_id, p_order_id, v_order.order_number,
    v_kind, v_title, v_body
  );
END;
$$;

-- GRANT: Only service_role (operator dashboard) may call transition_order.

GRANT EXECUTE ON FUNCTION transition_order(UUID, order_status) TO service_role;
GRANT EXECUTE ON FUNCTION insert_order_notification(UUID, UUID, TEXT, TEXT, TEXT, TEXT) TO service_role;

```typescript
// supabase/functions/send-fcm/index.ts
//
// Triggered by Supabase DB Webhook on INSERT into public.notifications.
// Reads the notification row, fetches all FCM tokens for the user,
// and sends a push via FCM HTTP v1 API.
//
// Required secret: FCM_SERVICE_ACCOUNT_JSON
//   Set in: Supabase Dashboard → Settings → Secrets
//   Value: JSON content of the Firebase service account private key file
//         (Firebase Console → Project settings → Service accounts → Generate new private key)
//
// Deploy: supabase functions deploy send-fcm
// DB Webhook: Table=notifications, Events=[INSERT], URL=this function URL,
//             Header: Authorization: Bearer <service_role_key>

import { createClient } from 'https://esm.sh/@supabase/supabase-js@2';

const FCM_ENDPOINT =
  'https://fcm.googleapis.com/v1/projects/qraft-e8a1d/messages:send';
const TOKEN_ENDPOINT = 'https://oauth2.googleapis.com/token';
const FCM_SCOPE = 'https://www.googleapis.com/auth/firebase.messaging';

// ---------------------------------------------------------------------------
// JWT helpers for FCM HTTP v1 OAuth2 service account auth
// ---------------------------------------------------------------------------

function base64UrlEncode(data: Uint8Array): string {
  return btoa(String.fromCharCode(...data))
    .replace(/\+/g, '-')
    .replace(/\//g, '_')
    .replace(/=+$/, '');
}

async function getAccessToken(serviceAccountJson: string): Promise<string> {
  const sa = JSON.parse(serviceAccountJson);
  const now = Math.floor(Date.now() / 1000);

  const header = { alg: 'RS256', typ: 'JWT' };
  const claim = {
    iss: sa.client_email,
    scope: FCM_SCOPE,
    aud: TOKEN_ENDPOINT,
    iat: now,
    exp: now + 3600,
  };

  const enc = new TextEncoder();
  const headerB64 = base64UrlEncode(enc.encode(JSON.stringify(header)));
  const claimB64 = base64UrlEncode(enc.encode(JSON.stringify(claim)));
  const signingInput = `${headerB64}.${claimB64}`;

  // Parse the PEM private key
  const pemContents = sa.private_key
    .replace(/-----BEGIN PRIVATE KEY-----/, '')
    .replace(/-----END PRIVATE KEY-----/, '')
    .replace(/\s/g, '');
  const binaryDer = Uint8Array.from(atob(pemContents), (c) => c.charCodeAt(0));

  const privateKey = await crypto.subtle.importKey(
    'pkcs8',
    binaryDer.buffer,
    { name: 'RSASSA-PKCS1-v1_5', hash: 'SHA-256' },
    false,
    ['sign']
  );

  const signature = await crypto.subtle.sign(
    'RSASSA-PKCS1-v1_5',
    privateKey,
    enc.encode(signingInput)
  );
  const jwt = `${signingInput}.${base64UrlEncode(new Uint8Array(signature))}`;

  const tokenRes = await fetch(TOKEN_ENDPOINT, {
    method: 'POST',
    headers: { 'Content-Type': 'application/x-www-form-urlencoded' },
    body: new URLSearchParams({
      grant_type: 'urn:ietf:params:oauth:grant-type:jwt-bearer',
      assertion: jwt,
    }),
  });
  const tokenData = await tokenRes.json();
  if (!tokenData.access_token) {
    throw new Error(`OAuth2 token error: ${JSON.stringify(tokenData)}`);
  }
  return tokenData.access_token as string;
}

// ---------------------------------------------------------------------------
// Handler
// ---------------------------------------------------------------------------

Deno.serve(async (req: Request) => {
  if (req.method !== 'POST') {
    return new Response('method not allowed', { status: 405 });
  }

  let body: Record<string, unknown>;
  try {
    body = await req.json();
  } catch {
    return new Response('invalid json', { status: 400 });
  }

  // Supabase DB webhook sends: { type: 'INSERT', record: <row>, ... }
  const record = body.record as Record<string, unknown> | undefined;
  if (!record) {
    return new Response('missing record', { status: 400 });
  }

  const notificationId = record.id as string;
  const userId = record.user_id as string;
  const titleEs = record.title_es as string;
  const bodyEs = record.body_es as string;
  const data = (record.data as Record<string, string>) ?? {};

  const supabase = createClient(
    Deno.env.get('SUPABASE_URL')!,
    Deno.env.get('SUPABASE_SERVICE_ROLE_KEY')!
  );

  // Fetch all FCM tokens for this user
  const { data: tokens, error: tokenErr } = await supabase
    .from('user_fcm_tokens')
    .select('token')
    .eq('user_id', userId);

  if (tokenErr || !tokens || tokens.length === 0) {
    console.log(`No FCM tokens for user ${userId}; skipping`);
    // Mark as sent (no tokens = nothing to do; not an error)
    await supabase
      .from('notifications')
      .update({ fcm_state: 'sent', sent_at: new Date().toISOString() })
      .eq('id', notificationId);
    return new Response('no_tokens', { status: 200 });
  }

  // Get OAuth2 access token
  const saJson = Deno.env.get('FCM_SERVICE_ACCOUNT_JSON');
  if (!saJson) {
    console.error('FCM_SERVICE_ACCOUNT_JSON not set');
    return new Response('server misconfigured', { status: 500 });
  }

  let accessToken: string;
  try {
    accessToken = await getAccessToken(saJson);
  } catch (e) {
    console.error('Failed to get FCM access token:', e);
    await supabase
      .from('notifications')
      .update({ fcm_state: 'failed', fcm_retries: (record.fcm_retries as number ?? 0) + 1 })
      .eq('id', notificationId);
    return new Response('fcm_auth_failed', { status: 500 });
  }

  // Send to each token
  let sentCount = 0;
  let failCount = 0;

  for (const { token } of tokens) {
    const payload = {
      message: {
        token,
        notification: { title: titleEs, body: bodyEs },
        data,  // contains order_id, order_number for deep-link
        android: {
          notification: {
            channel_id: 'qraft_orders',
            priority: 'HIGH',
          },
        },
        apns: {
          payload: {
            aps: { alert: { title: titleEs, body: bodyEs }, sound: 'default' },
          },
        },
      },
    };

    const fcmRes = await fetch(FCM_ENDPOINT, {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json',
        Authorization: `Bearer ${accessToken}`,
      },
      body: JSON.stringify(payload),
    });

    if (fcmRes.ok) {
      sentCount++;
    } else {
      const errBody = await fcmRes.text();
      console.error(`FCM send failed for token …${token.slice(-8)}: ${errBody}`);
      failCount++;
      // If token is invalid/unregistered, delete it to prevent future spam
      if (fcmRes.status === 404 || errBody.includes('UNREGISTERED')) {
        await supabase.from('user_fcm_tokens').delete().eq('token', token);
        console.log('Deleted unregistered token');
      }
    }
  }

  // Update notification state
  const finalState = failCount === 0 ? 'sent' : (sentCount === 0 ? 'failed' : 'sent');
  await supabase
    .from('notifications')
    .update({
      fcm_state: finalState,
      fcm_retries: failCount,
      sent_at: new Date().toISOString(),
    })
    .eq('id', notificationId);

  console.log(
    `send-fcm: notif ${notificationId} → sent=${sentCount} failed=${failCount}`
  );
  return new Response('ok', { status: 200 });
});
```
