```dart
import 'package:equatable/equatable.dart';

enum OrderStatus {
  paymentPending,
  paymentFailed,
  paid,
  receivedByOps,
  engraving,
  engravingFailed,
  engravingDone,
  shipped,
  delivered,
  cancelled,
  refunded,
}

extension OrderStatusX on OrderStatus {
  // Spanish display labels for the timeline widget
  String get label => switch (this) {
    OrderStatus.paymentPending  => 'Pago pendiente',
    OrderStatus.paymentFailed   => 'Pago fallido',
    OrderStatus.paid            => 'Pagado',
    OrderStatus.receivedByOps   => 'Recibido por ops',
    OrderStatus.engraving       => 'Grabando',
    OrderStatus.engravingFailed => 'Error de grabado',
    OrderStatus.engravingDone   => 'Grabado listo',
    OrderStatus.shipped         => 'Enviado',
    OrderStatus.delivered       => 'Entregado',
    OrderStatus.cancelled       => 'Cancelado',
    OrderStatus.refunded        => 'Reembolsado',
  };

  bool get isFinal => this == OrderStatus.cancelled || this == OrderStatus.refunded;

  // Used by timeline to determine which steps are "complete" vs "pending"
  int get stepIndex => OrderStatus.values.indexOf(this);

  static OrderStatus fromString(String raw) {
    const map = {
      'payment_pending'  : OrderStatus.paymentPending,
      'payment_failed'   : OrderStatus.paymentFailed,
      'paid'             : OrderStatus.paid,
      'received_by_ops'  : OrderStatus.receivedByOps,
      'engraving'        : OrderStatus.engraving,
      'engraving_failed' : OrderStatus.engravingFailed,
      'engraving_done'   : OrderStatus.engravingDone,
      'shipped'          : OrderStatus.shipped,
      'delivered'        : OrderStatus.delivered,
      'cancelled'        : OrderStatus.cancelled,
      'refunded'         : OrderStatus.refunded,
    };
    final result = map[raw];
    if (result == null) throw ArgumentError('Unknown OrderStatus: $raw');
    return result;
  }
}
```

```dart
class OrderSupabaseDataSource {
  final SupabaseClient _client;
  OrderSupabaseDataSource(this._client);

  // Paginated list: SELECT from orders WHERE user_id = auth.uid()
  // ORDER BY created_at DESC LIMIT limit OFFSET offset
  Future<List<Map<String, dynamic>>> getUserOrders({int offset = 0, int limit = 20}) async {
    final uid = _client.auth.currentUser!.id;
    final response = await _client
        .from('orders')
        .select()
        .eq('user_id', uid)
        .order('created_at', ascending: false)
        .range(offset, offset + limit - 1);
    return List<Map<String, dynamic>>.from(response as List);
  }

  // Realtime stream for a single order
  Stream<Map<String, dynamic>?> watchOrder(String orderId) {
    return _client
        .from('orders')
        .stream(primaryKey: ['id'])
        .eq('id', orderId)
        .map((rows) => rows.isNotEmpty ? rows.first : null);
  }

  // Order items (non-realtime — fetched once on detail load)
  Future<List<Map<String, dynamic>>> getOrderItems(String orderId) async {
    final response = await _client
        .from('order_items')
        .select()
        .eq('order_id', orderId);
    return List<Map<String, dynamic>>.from(response as List);
  }
}
```

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Flutter app → orders (SELECT) | RLS enforces user_id = auth.uid() for SELECT; client cannot read another user's orders |
| Flutter app → order_items (SELECT) | RLS uses EXISTS subquery; no direct mutation possible from client |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-10-05 | Information Disclosure | orders SELECT leak | mitigate | RLS SELECT policy `user_id = auth.uid()` prevents cross-user reads; StreamProvider uses authenticated client |
| T-10-06 | Tampering | order status mutation from client | mitigate | No UPDATE RLS policy for authenticated role; FSM enforced server-side; client is read-only |
| T-10-07 | Information Disclosure | order_items.qr_png_url exposed in UI | accept | PNG URLs are non-guessable UUIDs in private Storage bucket; RLS on order_items already restricts access to own orders |
| T-10-08 | Spoofing | orderId passed via navigation | mitigate | Supabase SELECT includes `.eq('user_id', uid)` in list; detail stream returns null if order belongs to another user (RLS rejects) |
</threat_model>
