-- Price unit: whole COP pesos (INT). E.g. 89000 = $89.000 COP.
-- Wompi requires amountInCents — multiply ×100 at checkout time (Phase 9).

  id TEXT PRIMARY KEY,              -- 'wood','acrylic','metal','leather','glass','stone'
  display_name_es TEXT NOT NULL,
  display_name_en TEXT NOT NULL,
  texture_url TEXT NOT NULL DEFAULT '',  -- public Supabase Storage URL; '' until assets uploaded
  laser_compatible BOOLEAN NOT NULL DEFAULT true,
  notes JSONB
);

-- products: base product catalog
CREATE TABLE IF NOT EXISTS products (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  slug TEXT UNIQUE NOT NULL,
  name_es TEXT NOT NULL,
  name_en TEXT NOT NULL,
  description_es TEXT,
  description_en TEXT,
  hero_image_url TEXT,
  is_active BOOLEAN NOT NULL DEFAULT true,
  created_at TIMESTAMPTZ NOT NULL DEFAULT NOW()

  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),

**6. `lib/features/marketplace/data/datasources/product_supabase_data_source.dart`**
Single denormalized query — no separate queries for products and materials:
```dart
import 'package:supabase_flutter/supabase_flutter.dart';
import '../models/product_variant_model.dart';

class ProductSupabaseDataSource {
  final SupabaseClient _supabase;
  const ProductSupabaseDataSource(this._supabase);

  Future<List<ProductVariantModel>> getActiveCatalog() async {
    final rows = await _supabase
        .from('product_variants')
        .select('''
          id, size_label, width_mm, height_mm, price_cop, sku, is_active,
          product:products(id, slug, name_es, name_en, description_es),
          material:materials(id, display_name_es, display_name_en, texture_url, laser_compatible)
        ''')
        .eq('is_active', true)
        .order('material_id')
        .order('price_cop');

    return (rows as List<dynamic>)
        .map((r) => ProductVariantModel.fromJson(r as Map<String, dynamic>))
        .toList();
  }
}
```

import 'package:flutter_riverpod/flutter_riverpod.dart';
import 'package:supabase_flutter/supabase_flutter.dart';
import '../../../marketplace/data/datasources/product_supabase_data_source.dart';
import '../../../marketplace/data/repositories/product_repository_impl.dart';
import '../../../marketplace/domain/entities/product_variant_entity.dart';
import '../../../marketplace/domain/repositories/product_repository.dart';

final productRepositoryProvider = Provider<ProductRepository>((ref) {
  final datasource = ProductSupabaseDataSource(Supabase.instance.client);
  return ProductRepositoryImpl(datasource);
});

/// Loads all active variants once; autoDispose clears cache when all listeners gone.
final catalogProvider = FutureProvider.autoDispose<List<ProductVariantEntity>>((ref) async {
  final repo = ref.read(productRepositoryProvider);
  return repo.getActiveCatalog();
});

/// Active material filter chip. Empty string = show all materials.
final activeMaterialFilterProvider = StateProvider.autoDispose<String>((ref) => '');

/// Derived: catalog filtered by active material (client-side, in-memory).
final filteredCatalogProvider = Provider.autoDispose<AsyncValue<List<ProductVariantEntity>>>((ref) {
  final catalog = ref.watch(catalogProvider);
  final filter = ref.watch(activeMaterialFilterProvider);
  if (filter.isEmpty) return catalog;
  return catalog.whenData(
    (variants) => variants.where((v) => v.material.id == filter).toList(),
  );
});
