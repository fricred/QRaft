3. `server_compute_order_totals` function:
```sql
CREATE OR REPLACE FUNCTION server_compute_order_totals(p_user_id UUID)
RETURNS TABLE (
  subtotal_cop INT,
  iva_cop      INT,
  envio_cop    INT,
  total_cop    INT
)
LANGUAGE plpgsql SECURITY DEFINER AS $$
DECLARE
  v_subtotal INT;
  v_iva      INT;
  v_envio    INT;
BEGIN
  SELECT COALESCE(SUM(pv.price_cop * ci.quantity), 0)
    INTO v_subtotal
    FROM cart_items ci
    JOIN product_variants pv ON pv.id = ci.product_variant_id
   WHERE ci.user_id = p_user_id;

  v_iva   := ROUND(v_subtotal * 0.19);
  v_envio := 15000;  -- flat rate; Phase 9 may introduce address-based rates

  RETURN QUERY SELECT v_subtotal, v_iva, v_envio, (v_subtotal + v_iva + v_envio);
END;
$$;
```

Note in a SQL comment that `price_cop` in `product_variants` stores whole COP pesos (NOT cents) — Wompi multiplication ×100 happens at checkout (Phase 9).

class CartItemEntity extends Equatable {
  final String id;

  final String? pngStorageUrl;  // null until snapshot plan runs

  final DateTime addedAt;

  const CartItemEntity({
    required this.id,
    required this.userId,
    required this.productVariantId,
    required this.qrCodeId,
    required this.customizationSnapshot,
    this.pngStorageUrl,
    required this.quantity,
    required this.addedAt,
  });

  Map<String, dynamic> toJson() => {
        'id': id,
        'user_id': userId,
        'product_variant_id': productVariantId,
        'qr_code_id': qrCodeId,
        'customization_snapshot': customizationSnapshot,
        'png_storage_url': pngStorageUrl,
        'quantity': quantity,
        'added_at': addedAt.toIso8601String(),

import 'package:equatable/equatable.dart';

/// Server-authoritative cart totals in whole COP pesos.
/// Populated exclusively from server_compute_order_totals() RPC.
/// Never computed client-side (Anti-Pattern 5 mitigation).
class CartTotalsEntity extends Equatable {
  final int subtotalCop;
  final int ivaCop;      // 19% of subtotalCop, rounded
  final int envioCop;    // flat 15000 COP (Phase 9 may use address-based)
  final int totalCop;    // subtotalCop + ivaCop + envioCop

  const CartTotalsEntity({
    required this.subtotalCop,
    required this.ivaCop,
    required this.envioCop,
    required this.totalCop,
  });

  factory CartTotalsEntity.fromRpcRow(Map<String, dynamic> row) =>
      CartTotalsEntity(
        subtotalCop: row['subtotal_cop'] as int,
        ivaCop: row['iva_cop'] as int,
        envioCop: row['envio_cop'] as int,
        totalCop: row['total_cop'] as int,
      );

  static const CartTotalsEntity zero = CartTotalsEntity(
    subtotalCop: 0,
    ivaCop: 0,
    envioCop: 15000,
    totalCop: 15000,
  );

  @override
  List<Object?> get props => [subtotalCop, ivaCop, envioCop, totalCop];
}

import '../entities/cart_item_entity.dart';
import '../entities/cart_totals_entity.dart';

abstract class CartRepository {
  /// Streams cart items for [userId] in real time via Supabase channel.
  Stream<List<CartItemEntity>> watchCart(String userId);

  /// Inserts a new cart item. If the (userId, productVariantId, qrCodeId)
  /// triple already exists, updates quantity instead (upsert semantics).
  Future<void> addItem({
    required String userId,
    required String productVariantId,
    required String qrCodeId,
    required Map<String, dynamic> customizationSnapshot,
    String? pngStorageUrl,
    int quantity = 1,
  });

  /// Updates the quantity of an existing cart item by [cartItemId].
  Future<void> updateQuantity(String cartItemId, int quantity);

  /// Removes a single item by [cartItemId].
  Future<void> removeItem(String cartItemId);

  /// Deletes all items for [userId].
  Future<void> clearCart(String userId);

  /// Calls server_compute_order_totals RPC.
  /// Returns CartTotalsEntity.zero when cart is empty.
  Future<CartTotalsEntity> computeTotals(String userId);
}

```dart
final cartRepositoryProvider = Provider<CartRepository>((ref) {
  return CartRepositoryImpl(CartSupabaseDataSource(Supabase.instance.client));
});

// Streams the current user's cart items.
// Returns empty list for unauthenticated users (authStateProvider == null).
final cartItemsProvider = StreamProvider<List<CartItemEntity>>((ref) {
  final user = ref.watch(authStateProvider);
  if (user == null) return const Stream.empty();
  return ref.read(cartRepositoryProvider).watchCart(user.id);
});

// Calls server_compute_order_totals RPC. Reruns when cartItemsProvider emits.
// Returns CartTotalsEntity.zero on empty cart or auth gap.
final cartTotalsProvider = FutureProvider<CartTotalsEntity>((ref) async {
  final cartAsync = ref.watch(cartItemsProvider);
  final user = ref.watch(authStateProvider);
  if (user == null) return CartTotalsEntity.zero;
  final items = cartAsync.value ?? [];
  if (items.isEmpty) return CartTotalsEntity.zero;
  return ref.read(cartRepositoryProvider).computeTotals(user.id);
});
```
