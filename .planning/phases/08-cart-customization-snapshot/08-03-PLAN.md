```dart
import 'dart:typed_data';
import 'package:pretty_qr_code/pretty_qr_code.dart';
import 'package:qr_flutter/qr_flutter.dart';

import '../../../qr_generator/domain/entities/qr_code_entity.dart';
import '../../../qr_generator/domain/services/qr_decoration_builder.dart';
import '../../../qr_generator/domain/services/qr_export_service.dart';

/// Result of a snapshot capture operation.
class SnapshotResult {
  /// Serialized QRCustomization + qr data payload for storage in cart_items.customization_snapshot.
  final Map<String, dynamic> customizationJson;

  /// 2160×2160 PNG bytes for upload to order-assets/ Storage bucket.
  final Uint8List pngBytes;

  const SnapshotResult({
    required this.customizationJson,
    required this.pngBytes,
  });
}

/// Freezes a [QRCodeEntity] into an immutable snapshot for cart/order storage.
///
/// The snapshot is a two-part artifact:
///   1. [customizationJson] — JSONB stored in cart_items.customization_snapshot;
///      contains all QRCustomization fields + the raw QR data string.
///   2. [pngBytes] — 2160×2160 print-safe PNG rasterized via QrExportService;
///      uploaded to Supabase Storage order-assets/ bucket.
///
/// Subsequent edits to the source QR in the library are irrelevant — this
/// snapshot is the single source of truth for the engraver (MKT-12).
class CustomizationSnapshotService {
  CustomizationSnapshotService._();

  /// Builds a [SnapshotResult] from [qrCode].
  ///
  /// The JSON snapshot merges all [QRCustomization] fields with
  /// {'data': qrCode.data} so the engraver always has the raw QR content.
  static Future<SnapshotResult> captureSnapshot(QRCodeEntity qrCode) async {
    final customizationJson = {
      ...qrCode.customization.toJson(),
      'data': qrCode.data,         // raw QR payload — belt-and-suspenders for engraver
      'display_data': qrCode.displayData,
      'qr_type': qrCode.type.name,
    };

    // Build QrImage using the frozen customization's error correction level.
    // For print-safe snapshots, EC-H is always appropriate (T-MKT-12-01).
    final qrImage = QrImage(
      data: qrCode.data,
      errorCorrectLevel: QrErrorCorrectLevel.H,
    );

    final decoration = QrDecorationBuilder.buildDecoration(qrCode.customization);

    final pngBytes = await QrExportService.exportPrintSafe(qrImage, decoration);

    return SnapshotResult(
      customizationJson: customizationJson,
      pngBytes: pngBytes,
    );
  }

  /// Builds only the JSON snapshot without rasterizing the PNG.
  /// Used for unit tests or offline fallback.
  static Map<String, dynamic> buildSnapshotJson(QRCodeEntity qrCode) => {
        ...qrCode.customization.toJson(),
        'data': qrCode.data,
        'display_data': qrCode.displayData,
        'qr_type': qrCode.type.name,
      };
}
```

**lib/features/cart/data/datasources/order_assets_storage_data_source.dart**

```dart
import 'dart:typed_data';
import 'package:supabase_flutter/supabase_flutter.dart';
import '../../domain/repositories/cart_repository.dart'; // for CartException

/// Handles upload of pre-rasterized QR PNGs to Supabase Storage bucket
/// 'order-assets/' (private bucket — access via signed URL).
///
/// Storage path convention: '{userId}/{qrCodeId}/{variantId}.png'
/// Bucket must exist and have RLS policy allowing authenticated users to
/// INSERT objects under their own userId prefix.
class OrderAssetsStorageDataSource {
  static const _bucket = 'order-assets';

  final SupabaseClient _client;

  const OrderAssetsStorageDataSource(this._client);

  /// Returns the storage path for a given combination.
  /// No leading slash; no bucket name prefix.
  static String storagePath(
    String userId,
    String qrCodeId,
    String variantId,
  ) =>
      '$userId/$qrCodeId/$variantId.png';

  /// Uploads [pngBytes] to order-assets/ and returns a signed URL valid for 1 hour.
  ///
  /// Throws [CartException('png_upload_failed', message)] on StorageException.
  Future<String> upload({
    required String userId,
    required String qrCodeId,
    required String variantId,
    required Uint8List pngBytes,
  }) async {
    final path = storagePath(userId, qrCodeId, variantId);
    try {
      await _client.storage.from(_bucket).uploadBinary(
            path,
            pngBytes,
            fileOptions: const FileOptions(
              contentType: 'image/png',
              upsert: true, // safe re-upload on retry
            ),
          );
      // Signed URL valid 1h — Phase 9 will re-generate at order creation.
      final signedUrl =
          await _client.storage.from(_bucket).createSignedUrl(path, 3600);
      return signedUrl;
    } on StorageException catch (e) {
      throw CartException('png_upload_failed', e.message);
    }
  }

  /// Writes the [pngStorageUrl] back to the cart_items row identified by [cartItemId].
  ///
  /// Called after a successful upload to patch the url without a full re-add.
  Future<void> patchPngUrl(String cartItemId, String pngStorageUrl) async {
    await _client
        .from('cart_items')
        .update({'png_storage_url': pngStorageUrl})
        .eq('id', cartItemId);
  }
}
```

```dart
Future<void> _addToCart(WidgetRef ref, BuildContext context) async {
  if (qrCodeId == null) {
    // guard: no QR selected
    ScaffoldMessenger.of(context).showSnackBar(
      const SnackBar(content: Text('Selecciona un QR para agregar al carrito')),
    );
    return;
  }
  final user = Supabase.instance.client.auth.currentUser;
  if (user == null) {
    ScaffoldMessenger.of(context).showSnackBar(
      const SnackBar(content: Text('Debes iniciar sesión')),
    );
    return;
  }

  setState(() => _isLoading = true);
  try {
    // 1. Find QRCodeEntity from already-loaded provider
    final qrCodes = ref.read(userQRCodesProvider).value ?? [];
    final qrCode = qrCodes.firstWhere(
      (q) => q.id == qrCodeId,
      orElse: () => throw CartException('qr_not_found', 'QR not found in library'),
    );

    // 2. Capture frozen snapshot (JSON + 2160px PNG)
    final snapshot = await CustomizationSnapshotService.captureSnapshot(qrCode);

    // 3. Add item to cart (pngStorageUrl null initially — patched below)
    final repo = ref.read(cartRepositoryProvider);
    await repo.addItem(
      userId: user.id,
      productVariantId: variant.id,
      qrCodeId: qrCodeId!,
      customizationSnapshot: snapshot.customizationJson,
    );

    // 4. Upload PNG to order-assets/ and patch the cart item's pngStorageUrl.
    //    Do this in background — cart item is already live, PNG is belt-and-suspenders.
    //    Failure here is non-fatal (JSONB snapshot is sufficient for engraver).
    unawaited(
      _uploadSnapshotPng(
        user: user,
        qrCodeId: qrCodeId!,
        variantId: variant.id,
        pngBytes: snapshot.pngBytes,
      ),
    );

    if (context.mounted) {
      Navigator.of(context).push(
        MaterialPageRoute(builder: (_) => const CartScreen()),
      );
    }
  } on CartException catch (e) {
    if (context.mounted) {
      ScaffoldMessenger.of(context).showSnackBar(
        SnackBar(content: Text('Error: ${e.message}')),
      );
    }
  } finally {
    if (mounted) setState(() => _isLoading = false);
  }
}

Future<void> _uploadSnapshotPng({
  required User user,
  required String qrCodeId,
  required String variantId,
  required Uint8List pngBytes,
}) async {
  try {
    final storageDs = OrderAssetsStorageDataSource(Supabase.instance.client);
    final pngUrl = await storageDs.upload(
      userId: user.id,
      qrCodeId: qrCodeId,
      variantId: variantId,
      pngBytes: pngBytes,
    );
    // Find the cart item just created and patch its pngStorageUrl
    await storageDs.patchPngUrl(
      // The UNIQUE constraint means we can look up by (user_id, product_variant_id, qr_code_id)
      // Use a direct Supabase query since CartRepository doesn't expose updatePngUrl
      // (keeping repository interface minimal per single-responsibility principle)
      // Supabase query approach:
      // await Supabase.instance.client.from('cart_items')
      //   .update({'png_storage_url': pngUrl})
      //   .eq('user_id', user.id)
      //   .eq('product_variant_id', variantId)
      //   .eq('qr_code_id', qrCodeId);
      // Since patchPngUrl takes a cartItemId, fetch it first:
      await _resolveCartItemId(user.id, variantId, qrCodeId),
      pngUrl,
    );
  } catch (_) {
    // Non-fatal: JSONB snapshot is the source of truth. Log in debug mode only.
    assert(() {
      // ignore: avoid_print
      print('[CartSnapshot] PNG upload failed: $_');
      return true;
    }());
  }
}

Future<String> _resolveCartItemId(
  String userId, String variantId, String qrCodeId) async {
  final row = await Supabase.instance.client
      .from('cart_items')
      .select('id')
      .eq('user_id', userId)
      .eq('product_variant_id', variantId)
      .eq('qr_code_id', qrCodeId)
      .single();
  return row['id'] as String;
}
```

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Flutter app → Supabase Storage order-assets/ | PNG bytes cross this boundary on upload |
| Flutter app → cart_items (png_storage_url patch) | Supabase update with RLS |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-08-03-01 | Tampering | PNG content in order-assets/ | accept | PNG is rasterized from QRCustomization + QrExportService with forced white bg; engraver validates visually; no executable content in PNG |
| T-08-03-02 | Information Disclosure | order-assets/ signed URLs | mitigate | Bucket is private; signed URLs expire in 1h; Phase 9 regenerates at order creation; users cannot access other users' paths (RLS on Storage INSERT path prefix = userId) |
| T-08-03-03 | Denial of Service | oversized PNG upload | accept | 2160px PNG via QrExportService is fixed-size (~300-800KB); no user-supplied dimensions; no DOS vector |
| T-08-03-04 | Tampering | pngStorageUrl patch via Supabase direct query | mitigate | RLS on cart_items UPDATE: user_id = auth.uid(); user can only patch their own cart items |
| T-08-03-05 | Tampering | customizationSnapshot JSONB content | accept | Server never executes JSON; it is opaque engraver data; no injection surface |
</threat_model>
