    // Add this static method to QrDecorationBuilder class:
    static dynamic buildBrush(QRCustomization c) {
      if (!c.hasGradient || c.gradientStart == null || c.gradientEnd == null) {
        return Color(c.foregroundColor);
      }
      final colors = [Color(c.gradientStart!), Color(c.gradientEnd!)];
      return switch (c.gradientType) {
        GradientType.linear => PrettyQrBrush.gradient(
            gradient: LinearGradient(
              begin: Alignment.topLeft,
              end: Alignment.bottomRight,
              colors: colors,
            ),
          ),
        GradientType.radial => PrettyQrBrush.gradient(
            gradient: RadialGradient(center: Alignment.center, colors: colors),
          ),
        GradientType.solid => Color(c.foregroundColor),
      };
    }
    ```

    import 'dart:typed_data';
    import 'dart:ui' as ui;
    import 'package:flutter/material.dart';
    import 'package:pretty_qr_code/pretty_qr_code.dart';

    class QrExportException implements Exception {
      final String code;
      QrExportException(this.code);
    }

    class QrExportService {
      QrExportService._();

      static Future&lt;Uint8List&gt; exportPrintSafe(
        QrImage qrImage,
        PrettyQrDecoration decoration,
      ) async {
        final exportDecoration = decoration.copyWith(background: Colors.white);
        final byteData = await qrImage.toImageAsBytes(
          size: 2160,
          format: ui.ImageByteFormat.png,
          decoration: exportDecoration,
        );
        if (byteData == null) throw QrExportException('export_failed');
        return byteData.buffer.asUint8List();
      }
    }
    ```
