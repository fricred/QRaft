    static PrettyQrShape buildShape(QRCustomization c) {

        final brush = buildBrush(c);
        final moduleSymbol = _moduleSymbol(c.modulePreset, brush);
        final eyeSymbol = _eyeSymbol(c.eyePreset, brush);

      } catch (_) {
        return const PrettyQrShape();
      }
    }
